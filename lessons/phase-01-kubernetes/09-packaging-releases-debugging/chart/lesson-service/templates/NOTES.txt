1. Wait for the release:
   kubectl -n {{ .Release.Namespace }} rollout status \
     deployment/{{ include "lesson-service.fullname" . }}

2. Forward the private ClusterIP Service:
   kubectl -n {{ .Release.Namespace }} port-forward \
     service/{{ include "lesson-service.fullname" . }} 18080:{{ .Values.service.port }}

3. In another terminal:
   curl -fsS http://127.0.0.1:18080/
